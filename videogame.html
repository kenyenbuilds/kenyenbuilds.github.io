function setup() {
createCanvas(600,600);
textAlign(CENTER);
textSize(20);
  
  
} // end setup

function draw() {
background(220);
if(gameState == "L1"){
  levelOne();
}
if(gameState == "L2"){
  levelTwo();
}
  if(gameState == "L3"){
    levelThree();
  }
  
    if(gameState == "youWin"){
    youWin();
  }
  
 text(("Score= " + score), width/2, 40);
} // end draw

function levelOne(){
  text("Level 1", width/2, height-20);
  var distToBall= dist(ballx, bally, mouseX, mouseY);
 if(distToBall= 5){
   gameState = "L2";
 }
 
  ellipse(ballx, bally, ballSize, ballSize);
  line(ballx, bally, mouseX, mouseY);
} //end of levelOne
